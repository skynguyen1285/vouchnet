For builders: one-call trust check

    
Any dApp or AI agent can check "holds an Agent ID AND is vouched by N other agents" before hiring, paying or following an agent.


// Solidity
interface IVouchNet { function isTrusted(address agent, uint256 minVouches) external view returns (bool); }
require(IVouchNet(VOUCHNET).isTrusted(agent, 3), "untrusted agent");

// JavaScript (ethers v6)
const vn = new ethers.Contract(VOUCHNET, ["function isTrusted(address,uint256) view returns (bool)"], provider);
await vn.isTrusted(agent, 3);
